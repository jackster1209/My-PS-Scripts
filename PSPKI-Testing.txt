Import-Module PSPKI

$CA = Get-CertificationAuthority `
    -ComputerName "your-ca.domain.local" |
    Where-Object Name -eq "YOUR-CA-NAME"

$Template = "YourCertificateTemplateName"

# Retrieve at most five issued certificate records
$Sample = @(
    Get-IssuedRequest `
        -CertificationAuthority $CA `
        -Filter @(
            "CertificateTemplate -eq $Template"
            "RequestID -gt 0"
        ) `
        -Page 1 `
        -PageSize 5
)

# Inspect the returned objects
$Sample | Select-Object `
    RequestID,
    CertificateTemplate,
    NotAfter,
    CommonName
